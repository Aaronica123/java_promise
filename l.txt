// Your class WITH listeners
class set_session {
    constructor(){
        this.auth = false;
        this.listeners = [];  // Empty phone list to start
    }
    
    // "Subscribe" means "Add me to the phone list"
    subscribe(listener) {
        this.listeners.push(listener);  // Add to list
        // Return a way to remove from list later
        return () => {
            this.listeners = this.listeners.filter(l => l !== listener);
        };
    }
    
    // "Notify" means "Call everyone on the phone list"
    notify() {
        this.listeners.forEach(listener => listener(this.auth));
    }
    
    check(bd){
        if(bd.state){
            this.auth = true;
            this.notify();  // 📞 CALL EVERYONE! "Hey, auth is now true"
        }
    }
    
    deleterole(){
        this.auth = false;
        this.notify();  // 📞 CALL EVERYONE! "Hey, auth is now false"
    }
}
